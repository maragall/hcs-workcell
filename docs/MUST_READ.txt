"""The gripper's force close, through the controller's Lua torque probe (lua/torque_probe, published in BioyondWorks).

While a Lua program runs the SDK's arm moves return OK and do nothing (2026-11-02), so the probe runs only for the close or the read: start it,
ask, sample, stop it. The clamp is MoveExtJWithForce (probe request 23): the jaw closes toward `command` under a torque limit and stops where it
meets the plate. A plate gives a torque at the limit and a width above the command; nothing gives the command's width. The clamp survives the probe
stopping: -0.50 torque and the same width after a 30 mm lift and 3 s (measured 2026-11-02, short edges at limit 0.5; 0.5 is the highest limit the
probe accepts). Each call opens its own short connection to the controller (BRCAPI port 56788).
"""
from __future__ import annotations

import time
from typing import Dict

from devices.controller import Brcapi, BrcapiError

PORT = 56788
IDX_W, IDX_NORM, IDX_BEAT = 5, 6, 7        # the probe's system variables: width x100, normalised torque x1000, heartbeat


class ClampError(Exception):
    pass


def _up(b: Brcapi) -> None:
    b.start_program()
    t0 = time.time()
    while True:
        if time.time() - t0 > 10.0:
            raise ClampError("the torque probe did not start")
        time.sleep(0.25)
        if time.time() - t0 < 1.0 or not b.state().lua_ready:
            continue
        beat = b.sysvars([IDX_BEAT]).get(IDX_BEAT, 0)
        time.sleep(0.4)
        if b.sysvars([IDX_BEAT]).get(IDX_BEAT, 0) != beat:
            return


def _sample(b: Brcapi, n: int = 10, dt: float = 0.1) -> Dict[str, float]:
    rows = []
    for _ in range(n):
        time.sleep(dt)
        v = b.sysvars(range(4, 9))
        rows.append((v.get(IDX_NORM, 0) / 1000.0, v.get(IDX_W, 0) / 100.0))
    t, w = [r[0] for r in rows], [r[1] for r in rows]
    return {"torque": round(sum(t) / n, 3), "torque_min": round(min(t), 3), "torque_max": round(max(t), 3), "width": round(sum(w) / n, 2)}


def _stop(b: Brcapi) -> None:
    try:
        b.stop_program()
    except Exception:
        pass
    time.sleep(0.7)
    b.close()


def close(ip: str, command: float, limit: float, speed: int = 20, port: int = PORT) -> Dict[str, float]:
    """Close toward `command` under the torque `limit` (0..0.5); the reading once it has stopped."""
    b = Brcapi(ip, port, log=None)
    b.connect()
    try:
        _up(b)
        try:
            code = b.run_job({0: 23, 1: int(round(command * 100)), 2: int(round(limit * 1000)), 3: int(speed)}, 15.0)
        except BrcapiError as e:
            code = "no push" if e.code == "no_push" else f"error {e.code}"      # BioyondWorks connected takes the push: the request clearing is the answer
        time.sleep(0.3)
        r = _sample(b)
        r["code"] = code
        return r
    finally:
        _stop(b)


def read(ip: str, port: int = PORT) -> Dict[str, float]:
    """The holding torque and width of the jaw as it is, nothing commanded."""
    b = Brcapi(ip, port, log=None)
    b.connect()
    try:
        _up(b)
        return _sample(b)
    finally:
        _stop(b)

    def clamp_holds(self, r: dict, command: float, limit: float, grip: str, near: bool = True):
        """None when the reading is a plate between the fingers, else the reason. The force is the signal: the torque is at the limit (a plate pushes
        back) and the jaw stopped above the command (the fingers did not reach it). The width only has to be near what this grip reads on the plate."""
        w, t = r["width"], abs(r["torque"])
        if t < 0.9 * limit:
            return f"torque {t:.2f} is under {0.9 * limit:.2f}: nothing presses back"
        if w < command + 0.3:
            return f"the jaw closed to {w:.2f}, the command {command}: nothing between the fingers"
        want = self.width_of(grip)
        if near and abs(w - want) > 1.0:
            return f"the pads stopped at {w:.2f}, this grip reads {want} on the plate: not the plate, or off its edge"
        return None

    def _clamp(self, grip: str, step: str):
        """Close under the torque limit (devices/clamp.py) and judge by force: returns (reading, None when a plate is held, else the reason)."""
        g = self.L["gripper"]
        limit, speed = float(g.get("clamp_limit", 0.5)), int(g.get("clamp_speed", 20))
        teaching = getattr(self, "teaching", False)          # setup: the plate's width is what is being learned: any width, a command well under it
        command = max(self.plate.jaw_closed[grip] - (2.0 if teaching else 0.0), 0.0)
        try:
            r = self.robot.clamp(command, limit, speed, self.checks.width_of(grip))
        except Exception as e:
            self._fault(step, "clamp_error", str(e))
            raise StopSafe(f"{step}: the force clamp failed: {e}")
        why = self.checks.clamp_holds(r, command, limit, grip, near=not teaching)
        self.tele.log("jaw", "clamp", why or "holding", grip=grip, width=r["width"], torque=r["torque"], limit=limit)
        self._observe(f"jaw:clamp:{grip}", r["width"])
        return r, why

    def _carry_check(self, step: str, grip: str) -> None:
        """The plate is still pressed between the fingers: the torque read back is at the limit and the width has not moved."""
        g = self.L["gripper"]
        limit = float(g.get("clamp_limit", 0.5))
        try:
            r = self.robot.clamp_read()
        except Exception as e:
            self._fault(step, "clamp_error", str(e))
            raise StopSafe(f"{step}: the force read failed: {e}")
        why = self.checks.clamp_holds(r, self.plate.jaw_closed[grip], limit, grip)
        self.tele.log("jaw", "carry", why or "holding", grip=grip, width=r["width"], torque=r["torque"])
        if why:
            self._fault(step, "plate_lost", why)
            raise StopSafe(f"{step}: the plate is not in the claw: {why}")

        self._health()
        hover = offset(pose, [0, 0, fixture["hover_mm"]])
        if trajectory_in:
            self._enter(trajectory_in, False, hover, fixture_name)

        target = self._seat(pose, [0.0] * 6)

        self._carry_check(step, grip)
        self.store.done(plate_at="gripper", grip=grip)

    def _place(self, fixture_name: str, pose: List[float], fixture: dict, at: str, trajectory_in: Optional[str],

    def _steps(self, slot: str):
        """The two routes of one plate: microscope -> hotel slot (out) and hotel slot -> microscope (back).
        The microscope's route to the stage is fixtures.<scope>.trajectory."""
        scope, F = self.scope, self.L["fixtures"]
        P0, P90 = F["polar"]["pose_at_0"]["pose"], F["polar"]["pose_at_90"]["pose"]
        route = F[scope].get("trajectory", "microscope_to_polar")
        out = [
            (f"pick:{scope}", lambda: self._pick(scope, F[scope]["pose"], F[scope], "short_edges", None)),
            ("place:polar", lambda: self._place("polar", P0, F["polar"], "polar", route, gate="polar@0")),
            ("polar:90", lambda: self._polar_rotate(90, P0)),
            ("pick:polar", lambda: self._pick("polar", P90, F["polar"], "long_edges", None, gate="polar@90")),
            (f"place:hotel:{slot}", lambda: self._hotel_place_free(slot)),
        ]
        back = [
            (f"pick:hotel:{slot}", lambda: self._hotel_pick(slot)),
            ("place:polar", lambda: self._place("polar", P90, F["polar"], "polar", None, gate="polar@90")),
            ("polar:0", lambda: self._polar_rotate(0, P90)),
            ("pick:polar", lambda: self._pick("polar", P0, F["polar"], "short_edges", None, gate="polar@0")),
            (f"place:{scope}", lambda: self._place(scope, F[scope]["pose"], F[scope], scope, route, reverse=True)),
        ]
        return out, back

    def _hotel_mouth(self, approach: List[float], holding: Optional[str] = "now") -> List[float]:
        """Where the claw (the plate in it, or the fingers) is just outside the rack, in line with an approach point: the wrist
        backed along the slot until the front of what the claw carries is clear of the hotel's first part. From the geometry
        (the parts of volumes.obstacles.hotel and the footprint of the claw with what it holds), not a distance chosen by hand.
        Never beyond the approach point itself (an empty claw is outside the rack there already)."""
        from routes.volumes import Volumes
        if holding == "now":
            grip = getattr(getattr(self, "store", None), "s", None) and self.store.s.grip
            holding = grip if grip and grip != "none" else None
        v = Volumes(self.L, self.safety, holding)
        h = v.obstacles["hotel"]
        front = min(q["x"][0] - q.get("margin_mm", 1.0) for q in h["parts"]) if h.get("parts") else h["x"][0]
        ahead = max(r[0][1] for r in v.footprint)
        return [min(approach[0], front - ahead - MOUTH_CLEAR_MM)] + list(approach[1:])

    def _hotel_go_to_approach(self, slot: str, step: str):
        """Route A end -> junction -> (retreat, climb) -> approach point of the slot, all at carry height. Planned: from
        wherever the arm is straight to the approach point, around the hotel."""
        prof = self.safety.profile
        _, approach, _, _ = self._hotel_geometry(slot)
        if self._planned():
            if max(abs(self.robot.pose()[i] - approach[i]) for i in range(3)) <= 5.0:     # resuming at the approach point
                self.resuming = False
                return
            here = self.robot.pose()
            mouth = self._hotel_mouth(approach)
            if mouth[0] < approach[0] - 1.0 and math.hypot(here[0] - approach[0], here[1] - approach[1]) <= 8.0 and abs(here[2] - approach[2]) > 5.0:
                self._hotel_row_change(slot, step, approach)
                return
            if mouth[0] >= approach[0] - 1.0:
                self.go_to(approach, f"hotel approach {slot}")
                return
            # The claw turns, and the plate in it swings, outside the rack, at the height of the slot's approach: one gap between two
            # trays holds one plate, so a turn at any other height with the plate's front in the rack hits a tray. From the
            # mouth the way in is straight along the slot.
            self.go_to(mouth, f"hotel mouth {slot}")
            self._hotel_leg(mouth, approach, f"{step} approach")
            return
        foot, top = self._hotel_climb(slot)
        junction = self._hotel_junction(slot)
        here = self.robot.pose()
        if max(abs(here[i] - approach[i]) for i in range(3)) <= 5.0:     # resuming at the approach point
            self.resuming = False
            return
        if max(abs(here[i] - junction[i]) for i in range(3)) > 5.0:
            self._waypoints("polar_to_hotel")
            self._move("J", junction, f"{step} junction", prof.approach)
        if foot is not None:
            self._move("L", foot, f"{step} retreat", prof.approach)
            self._move("L", top, f"{step} climb", prof.approach)
            self._move("L", approach, f"{step} approach", prof.approach)

    def _square_up(self, target: List[float], label: str) -> None:
        """Before a slide into a rack: the claw square to it and on its line. A joint move into the approach point leaves a yaw error
        (-0.49 deg at B7, 2026-11-02 01:10) that nothing checked, and the plate is 85.1 mm in an 88 mm opening between the posts: 0.49 deg
        swings its far corner 2.2 mm, against 1.4 mm to spare. One slow straight move to the exact pose; refuse the slide if it is still off."""
        for _ in range(2):
            here = self.robot.pose()
            dyaw = abs((here[5] - target[5] + 180.0) % 360.0 - 180.0)
            dxy = max(abs(here[0] - target[0]), abs(here[1] - target[1]))
            if dyaw <= 0.1 and dxy <= 0.5:
                return
            self._move("L", list(target), f"{label} square up (yaw off {dyaw:.2f} deg, {dxy:.1f} mm)", self.safety.profile.contact)
        here = self.robot.pose()
        dyaw = abs((here[5] - target[5] + 180.0) % 360.0 - 180.0)
        if dyaw > 0.25:
            raise StopSafe(f"{label}: the claw is {dyaw:.2f} deg off square to the rack after squaring up; the plate would meet a post")

def layout_sig(layout: dict) -> str:
    """A hash of the geometry a route depends on: the volumes, the regions, the routes block, and of every fixture only the keys that place
    or size it. Flags (a plate on the module), widths, the occupancy, the setup notes and the vision settings are left out: the first
    version of this hash held them, changed every move, and no stored leg ever matched (2026-11-02)."""
    fixtures = {name: {k: v for k, v in fx.items() if k in GEOMETRY_KEYS} for name, fx in layout.get("fixtures", {}).items() if isinstance(fx, dict)}
    geometry = [layout.get("volumes"), layout.get("regions"), layout.get("routes"), fixtures]
    return hashlib.sha1(json.dumps(geometry, sort_keys=True, default=str).encode()).hexdigest()

    def grips(self) -> Section:
        c = self.cell
        widths: dict = {}

        def poses():
            F = c.layout()["fixtures"]["polar"]
            return F, F["pose_at_0"]["pose"], F["pose_at_90"]["pose"]

        def corrected(p0):
            dx, dy, dyaw = self.found.get("plate_offset") or c.layout().get("setup", {}).get("plate_offset") or (0, 0, 0)
            p = offset(p0, [dx, dy, 0.0])
            p[5] = p0[5] + dyaw
            return p

        def to_module():
            """The plate is in the holder (the seat was just proved): microscope to the module along the microscope's taught way and the planner."""
            self.ready()
            cyc = self.cyc()
            st = cyc.store.s
            st.plate_at, st.grip, st.step_done = self.scope, "none", True
            cyc.store.save()
            steps = cyc._steps("A1")[0][:2]                       # pick:microscope, place:polar (the slot's steps are not run)
            self.guard(lambda: cyc._run_steps(steps))
            return {"plate": "on the module at 0 deg"}

        def short():
            self.ready()
            F, p0, _ = poses()
            cyc = self.cyc()
            self.leave_holder(release=True)                      # a plate may still be in the fingers at the microscope seat
            self.guard(lambda: cyc.move_to("polar.pose_at_0"))
            if cyc.polar.driver == "ext_axis" and abs(((cyc.polar.read_angle() + 180.0) % 360.0) - 180.0) > 3.0:
                print(f"   the module reads {cyc.polar.read_angle():.1f} deg: turning it to the short-edge grip (0) first")
                self.guard(lambda: cyc._polar_rotate(0, p0))
                self.guard(lambda: cyc.move_to("polar.pose_at_0"))
            cyc.teaching = True                                     # the first clamp on this plate: any width, the force says whether it is held
            try:
                self.guard(lambda: cyc._pick("polar", corrected(p0), F, "short_edges", None, gate="polar@0"))
            finally:
                cyc.teaching = False
            widths["short_edges"] = round(c.robot.jaw(), 2)
            return {"short_closed": widths["short_edges"]}

        def grip_offset():
            size = self.found.get("hotel_size") or c.layout().get("setup", {}).get("hotel", {}).get("size")
            back = c.layout()["fixtures"]["hotel"].get("grip_back_mm", {}).get(str(size))
            if back is None:
                raise Refused(f"the {size}-slot hotel has no grip_back_mm in layout.json (fixtures.hotel). Measure how far "
                              "back the plate is gripped for it with the caliper and enter it")
            self.found["grip_offset"] = back
            return {"grip_offset": back}

        def long():
            F, p0, p90 = poses()
            cyc = self.cyc()
            self.guard(lambda: cyc._place("polar", corrected(p0), F, "polar", None, gate="polar@0"))
            self.guard(lambda: cyc._polar_rotate(90, p0))
            cyc.teaching = True
            try:
                self.guard(lambda: cyc._pick("polar", p90, F, "long_edges", None, gate="polar@90"))
            finally:
                cyc.teaching = False
            widths["long_edges"] = round(c.robot.jaw(), 2)
            return {"long_closed": widths["long_edges"]}

        def band():
            def write(L):
                g = L["gripper"]
                for grip, w in widths.items():
                    g.setdefault("plate_width", {})[grip] = w                 # where the pads stop on this plate under the force limit
                    g["closed"][grip] = round(w - 1.0, 2)                     # the clamp command: 1 mm under it
                    g["accept_band"][grip] = geometry.accept_band(w, BAND_BELOW, BAND_ABOVE[grip])
                self.setup_node(L).update(grips={"short_closed": widths["short_edges"], "long_closed": widths["long_edges"],
                                                 "grip_offset": self.found["grip_offset"], "measured_at": _stamp()})
            self.edit(write)
            cyc = self.cyc()
            for grip, w in widths.items():                            # the cycle in hand judges the next picks against what was just learned
                cyc.checks.width[grip] = w
                cyc.checks.closed[grip] = cyc.plate.jaw_closed[grip] = round(w - 1.0, 2)
            g = c.layout()["gripper"]["accept_band"]
            return {"accept_band": {k: g[k] for k in widths}, "measured_at": _stamp()}

        def reseat():
            F, p0, p90 = poses()
            cyc = self.cyc()
            seen = []
            for _ in range(RESEATS):
                self.guard(lambda: cyc._place("polar", p90, F, "polar", None, gate="polar@90"))
                self.guard(lambda: cyc._pick("polar", p90, F, "long_edges", None, gate="polar@90"))
                seen.append(round(c.robot.jaw(), 2))
            lo, hi = c.layout()["gripper"]["accept_band"]["long_edges"]
            bad = [w for w in seen if not lo <= w <= hi]
            if bad:
                raise Refused(f"re-grip widths {seen} left the band {lo} to {hi}: the plate does not seat the same way each time")
            return {"re-grip widths": seen, "spread": geometry.spread(seen)}

            F, p0, p90 = poses()

            self.guard(lambda: cyc._place("polar", p90, F, "polar", None, gate="polar@90"))
            self.guard(lambda: cyc._polar_rotate(0, p90))

            """Back to the microscope's seat: the two legs between the microscope and the module are now done once each."""
            cyc = self.cyc()
            steps = cyc._steps("A1")[1][3:]                          # pick:polar by the short edges, place:microscope

            return {"plate": "on the microscope stage"}

            Step("polar", "place plate back on the rotation module at 0 deg", put_back, moves=True),
            Step("polar", "carry the plate back to the microscope", to_scope, moves=True)])

"""
Keeps `cli.py shuffle` running for days. When the shuffle stops with a refusal or a fault, the reason is logged and it starts again
(the shuffle's own start takes the plate it finds in the claw, on the module or on the stage); the wait before the next start grows
with each failure in a row. It gives up, and leaves the arm where it is, when the same reason stops it GIVE_UP times in a row without
a run of at least GOOD_RUN_S in between: a repeat fault is a rule to write, not something to retry all night.

    python tools/keep_running.py [--speed supervised]        stop: create runs/STOP (it ends between plates, then this ends)
"""
import argparse
import os
import re
import subprocess
import sys
import time

HERE = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
RUNS = os.path.join(HERE, "runs")
LOG = os.path.join(RUNS, "keep_running.log")
GIVE_UP = 4                # the same reason this many times in a row
GOOD_RUN_S = 900.0
BACKOFF_S = (10, 20, 30, 45)     # short: a stopped arm that holds a plate lost it within 5 minutes of the last command (2026-11-02, the cause is probably the controller dropping the servos); every restart is a new command


def log(msg: str) -> None:
    line = f"{time.strftime('%Y-%m-%d %H:%M:%S')}  {msg}"
    print(line, flush=True)
    with open(LOG, "a", encoding="utf-8") as f:
        f.write(line + "\n")


def reason_of(path: str) -> str:
    """What the run said when it stopped: the REFUSED / fault line, else the last non-empty line."""
    try:
        text = open(path, encoding="utf-8", errors="replace").read().splitlines()
    except OSError:
        return "no output"
    for line in reversed(text):                # the refusal first: the SDK may crash with a .NET trace after it, which says nothing
        if "REFUSED" in line:
            return re.sub(r"\d\d:\d\d:\d\d", "", line).strip()[:200]
    for line in reversed(text[-60:]):
        if "Traceback" in line or "Error" in line or "Exception" in line:
            return re.sub(r"\d\d:\d\d:\d\d", "", line).strip()[:200]
    for line in reversed(text):
        if line.strip():
            return re.sub(r"\d\d:\d\d:\d\d", "", line).strip()[:200]
    return "no output"


def main(argv=None) -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--speed", default="supervised")
    a = ap.parse_args(argv)
    os.makedirs(RUNS, exist_ok=True)
    stop = os.path.join(RUNS, "STOP")
    last, repeats, fails = None, 0, 0
    env = dict(os.environ, PYTHONIOENCODING="utf-8")
    log(f"keep_running starts: shuffle at {a.speed}; stop with {stop}")
    while True:
        if os.path.exists(stop):
            log("runs/STOP exists: not starting another run")
            return 0
        out = os.path.join(RUNS, f"keep_running_{time.strftime('%Y%m%d_%H%M%S')}.txt")
        t0 = time.time()
        with open(out, "w", encoding="utf-8") as f:
            code = subprocess.call([sys.executable, "-X", "utf8", "-u", "cli.py", "shuffle", "--speed", a.speed],
                                   cwd=HERE, stdout=f, stderr=subprocess.STDOUT, env=env)
        ran = time.time() - t0
        if os.path.exists(stop):
            log(f"stopped by runs/STOP after {ran / 60:.1f} min (exit {code})")
            return 0
        why = reason_of(out)
        if ran >= GOOD_RUN_S:
            repeats, fails = 0, 0
        key = re.sub(r"[\d.\-]+", "#", why)[:80]
        repeats = repeats + 1 if key == last else 1
        last, fails = key, fails + 1
        log(f"run ended after {ran / 60:.1f} min (exit {code}): {why}   [same reason {repeats}x, failures in a row {fails}; output {out}]")
        if repeats >= GIVE_UP:
            msg = f"gave up: the same reason stopped the shuffle {repeats} times in a row: {why}"
            log(msg)
            with open(os.path.join(RUNS, "SUPERVISOR_GAVE_UP.txt"), "w", encoding="utf-8") as f:
                f.write(msg + "\n")
            return 1
        wait = BACKOFF_S[min(fails, len(BACKOFF_S)) - 1]
        log(f"waiting {wait} s, then the shuffle starts again")
        for _ in range(wait):
            if os.path.exists(stop):
                break
            time.sleep(1)


if __name__ == "__main__":
    sys.exit(main())
